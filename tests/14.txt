program TestProgram;

var
    x, y, counter: integer;
    price, total: real;
    finished: boolean;

begin
    { Read values }
    read(x);
    read(y);

    { Integer arithmetic }
    counter := 0;
    counter := x + y;
    counter := counter * 2;
    counter := counter - 5;
    counter := counter div 2;

    { Real arithmetic }
    price := 12.5;
    total := price * 3.0;
    total := total + 7.5;
    total := total / 2.0;

    { Write expressions }
    write(x);
    write(y);
    write(counter);
    write(price);
    write(total);
    write(x + y * 2);

    { If-then-else }
    if x > y then
        write(x)
    else
        write(y);

    { Boolean conditions }
    if (x > 0) and (y > 0) then
        write(1)
    else
        write(0);

    if (x = y) or (x < 0) then
        write(100)
    else
        write(200);

    { While loop }
    counter := 0;

    while counter < 5 do
    begin
        write(counter);
        counter := counter + 1;
    end;

    { Nested if statements }
    if x >= 10 then
    begin
        if y <= 20 then
            write(x + y)
        else
            write(x - y);
    end
    else
        write(0);

    { More arithmetic }
    total := (price + 2.5) * (3.0 - 1.0);
    write(total);

    { Boolean variable assignment }
    finished := true;

    if finished then
        write(999)
    else
        write(0);

end.
